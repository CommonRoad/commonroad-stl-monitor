\subsection{Velocity}
\label{sec:speed_limit}


Speed limit provided by traffic regulations for a vehicle $V_i$:
\begin{align}
\speedLimitLane := \sup \big( \big\{v   \, \big| \, \forall l \in \mathtt{\occupiedLaneletsFunction}(\mathbb{L}, V_i): v = \speedLimitFunction(l) \big\} \big)
\end{align}

Required speed limit provided by traffic regulations for a vehicle $V_i$:
\begin{align}
\requiredSpeedLane := \sup \big( \big\{v   \, \big| \, \forall l \in \mathtt{\occupiedLaneletsFunction}(\mathbb{L}, V_i): v = \requiredSpeedFunction(l) \big\} \big)
\end{align}

The ego vehicle must always be able to stop within the field of view. Therefore, the maximum allowed velocity based on the field of view $v_{sr}$ is chosen such that
\begin{align}
\speedLimitFOV := 
\begin{split}
& \mathrm{proj}(\xi(t_{s}(\tilde{u}_{brake}(\cdot));x_\mathrm{lon}^\mathrm{sr},\tilde{u}_{brake}(\cdot))) \leq fov, \\
& x_\mathrm{lon}^\mathrm{sr} = [0, \speedLimitFOV, \overline{a}, \overline{j}]^T.
\end{split}
\end{align}

The ego vehicle must always be able reduce its velocity to a upcoming speed limit such that it does not brake abruptly. 
Therefore, the maximum allowed velocity must be limited such that the braking trajectory does not exceed acceleration and jerk thresholds.
\begin{align}
\begin{split}
& \speedLimitBraking :=  \sup \big( \big\{v   \, \big| \, x_\mathrm{f} = \xi(t^*;x_\mathrm{0}, u(\cdot))) \land \forall t \in [t_0, t^*]:\aEgo(t) \geq a^* \land \jEgo(t) \geq j^* \big\} \big) \\
& x_\mathrm{0} = [0,\speedLimitBraking, \aMaxEgo, \jMaxEgo]^T. \\
& x_\mathrm{f} = [fov, v_\mathrm{sl}^*, 0 , 0]^T,
\end{split}
\end{align}
where $v_\mathrm{sl}^*$ is the maximum speed limit which is allowed after free driving, e.g., in Germany if there is now active speed limit, the next speed limit is maximal $100$~km/h.
The velocity limits $\speedLimitFOV$ and $\speedLimitBraking$ can be calculated offline using binary search. The field of view based velocity limit $\speedLimitFOV$ can change during runtime, e.g., due to changing weather conditions.

\begin{align}
\laneSpeedLimitPredicate \iff \vEgo \leq \speedLimitLane
\end{align}

\begin{align}
\fovSpeedLimitPredicate \iff \vEgo \leq \speedLimitFOV
\end{align}

\begin{align}
\brakingSpeedLimitPredicate \iff \vEgo \leq \speedLimitBraking
\end{align}

\begin{align}
\roadConditionSpeedLimitPredicate \iff \vEgo \leq \speedLimitWeather
\end{align}


\begin{align}
\minSpeedLimitPredicate \iff \requiredSpeedLane < \speedLimitFOV \land \requiredSpeedLane < \speedLimitWeather  \implies \requiredSpeedLane \leq \vEgo
\end{align}

\begin{align}
\preservesTrafficFlowPredicate \iff \neg \slowLeadingVehicle \implies \vEgo - \velocityLeadingVehicleFunction() > \deltaVTrafficFlow
\end{align}

\subsection{Safe distance}

\begin{equation*} 
\safeDistance(\vrear, \vlead) =
\begin{cases}
\safeDistanceA(\vrear, \vlead), & \big(\aminlead > \aminrear \big) \land \big(\vleadStar < \vrear \big) \land \\
& \big(\timeDelay \leq \frac{\vlead}{-\aminlead}\big) \land \big(\frac{\vrear}{\aminrear} > \frac{\vleadStar}{\aminlead}\big), \\
\safeDistanceB(\vrear, \vlead), & \text{otherwise},
\end{cases}
\end{equation*}
where
\allowdisplaybreaks
\begin{align*}
\vleadStar = & \; \begin{cases}
\vlead - \aminlead \timeDelay, & \timeDelay \leq \frac{\vlead(t_k)}{-\aminlead}, \\
0, & \text{otherwise}, 
\end{cases} \\
\safeDistanceA(\vrear, \vlead) := & \; \frac{\big(\vlead \! - \! \aminlead \timeDelay \! - \! \vrear^2\big)^2}{2 (\aminlead - \aminrear)} \! + \! \vrear \timeDelay \; - \vlead \timeDelay - \frac{1}{2} \aminlead \timeDelay^2 , \\
\safeDistanceB(\vrear, \vlead) := & \; \frac{\vlead^2}{2 \aminlead} - \frac{\vrear^2}{2 \aminrear} + \vrear \timeDelay ,  \\
\end{align*} 
and \timeDelay denotes the reaction time of the ego vehicle.


\begin{align}
\safeDistanceFrontPredicate \iff \safeDistance(\vEgo, \vOther) \leq (\rearPositionFunction(\otherVehicle) - \frontPositionFunction(V_\mathrm{ego}))
\end{align}

\begin{align}
\safeDistanceRearPredicate \iff \safeDistance(\vOther, \vEgo) \leq (\rearPositionFunction(\egoVehicle) - \frontPositionFunction(\otherVehicle))
\end{align}

\subsection{Unnecessary braking}
\begin{align}
\begin{split}
&\mathtt{unnecessary\_braking} \iff \\ 
& a_\mathrm{ego} < 0 \land j_\mathrm{ego} < j_\mathrm{abrupt} \land \nexists \ V' \in \ \mathbb{V} : \vehiclesInFrontOfEgoVehicleFunction = V' \land \speedLimitLane \neq \emptyset  \\ 
& \lor \exists V' \in \mathbb{V} : \vehiclesInFrontOfEgoVehicleFunction = V' \land a_\mathrm{ego}  <  a_\mathrm{abrupt} \land j_\mathrm{ego}  <  j_\mathrm{abrupt} \\ 
& \lor  \big(\speedLimitLane \neq \emptyset  \land a_\mathrm{ego}  <  a_\mathrm{abrupt} \land j_\mathrm{ego}  <  j_\mathrm{abrupt}\big)
\end{split}
\end{align}
